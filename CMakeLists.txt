cmake_minimum_required(VERSION 3.24)
project(SunglassesCompute LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT WIN32)
    message(FATAL_ERROR "The D3D11 reference runner requires Windows")
endif()

option(SUNGLASSES_BUILD_HEADLESS_CHECKS "Build headless worker integration probe" ON)
if(SUNGLASSES_BUILD_HEADLESS_CHECKS)
    add_executable(sunglasses-worker-probe tests/native_worker_probe.cpp native/desktop/worker_process.cpp)
    target_compile_features(sunglasses-worker-probe PRIVATE cxx_std_20)
    target_compile_definitions(sunglasses-worker-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(sunglasses-worker-probe PRIVATE /W4 /WX /utf-8 /EHsc)
    endif()
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/highlight.hlsl" HIGHLIGHT_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/highlight.hlsl)
configure_file(native/shader_source.h.in generated/shader_source.h @ONLY)
add_executable(sunglasses-compute native/compute.cpp)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/automatic.hlsl" AUTOMATIC_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/automatic.hlsl)
configure_file(native/automatic_source.h.in generated/automatic_source.h @ONLY)
add_library(sunglasses-auto-core STATIC native/automatic.cpp native/session.cpp)
add_executable(sunglasses-auto-compute native/auto_compute.cpp)
target_link_libraries(sunglasses-auto-compute PRIVATE sunglasses-auto-core)
foreach(target sunglasses-compute sunglasses-auto-core sunglasses-auto-compute)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(${target} PRIVATE d3d11 d3dcompiler)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
    endif()
endforeach()

option(SUNGLASSES_BUILD_DESKTOP "Build the native offline desktop host" ON)
if(SUNGLASSES_BUILD_DESKTOP)
    enable_language(RC)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(STRINGS_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_strings.h")
    add_custom_command(OUTPUT "${STRINGS_HEADER}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_strings.py" "${STRINGS_HEADER}"
        DEPENDS scripts/generate_strings.py sunglasses/locales/en.json sunglasses/locales/zh-CN.json
        VERBATIM)
    # Let CMake merge/embed the manifest for both incremental and full links.
    add_executable(sunglasses-desktop WIN32 native/desktop/app.cpp native/desktop/worker_process.cpp
        native/desktop/app.rc native/desktop/app.manifest "${STRINGS_HEADER}")
    set_target_properties(sunglasses-desktop PROPERTIES OUTPUT_NAME SunglassesLab)
    target_compile_features(sunglasses-desktop PRIVATE cxx_std_20)
    target_compile_definitions(sunglasses-desktop PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_include_directories(sunglasses-desktop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(sunglasses-desktop PRIVATE user32 gdi32 comctl32 ole32 shell32 uuid)
    if(MSVC)
        target_compile_options(sunglasses-desktop PRIVATE /W4 /WX /utf-8 /EHsc)
        set_property(TARGET sunglasses-desktop PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()

    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/screen.hlsl" SCREEN_SOURCE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/screen.hlsl)
    configure_file(native/screen_source.h.in generated/screen_source.h @ONLY)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/apple_vfm.hlsl" APPLE_VFM_SOURCE)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sunglasses/data/apple_ufm_kernels_6e9604d.txt" APPLE_VFM_KERNELS)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/sunglasses/data/apple_ufm_kernels_6e9604d.txt" APPLE_VFM_KERNEL_HASH)
    if(NOT APPLE_VFM_KERNEL_HASH STREQUAL "209bc2e47a8a8dcaab6172787dc2721efe38a6b5d9d49fdeecd5938100a8dc79")
        message(FATAL_ERROR "Apple VFM kernel data differs from the pinned reference")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        shaders/apple_vfm.hlsl sunglasses/data/apple_ufm_kernels_6e9604d.txt)
    configure_file(native/apple_vfm_source.h.in generated/apple_vfm_source.h @ONLY)
    add_library(sunglasses-screen-core STATIC native/screen/pipeline.cpp native/screen/apple_vfm.cpp)
    target_compile_features(sunglasses-screen-core PRIVATE cxx_std_20)
    target_compile_definitions(sunglasses-screen-core PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(sunglasses-screen-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(sunglasses-screen-core PUBLIC sunglasses-auto-core d3d11 d3dcompiler)
    if(SUNGLASSES_BUILD_HEADLESS_CHECKS)
        add_executable(sunglasses-screen-probe tests/native_screen_probe.cpp)
        target_compile_features(sunglasses-screen-probe PRIVATE cxx_std_20)
        target_compile_definitions(sunglasses-screen-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(sunglasses-screen-probe PRIVATE sunglasses-screen-core)
        if(MSVC)
            target_compile_options(sunglasses-screen-probe PRIVATE /W4 /WX /utf-8 /EHsc)
        endif()
    endif()
    add_executable(sunglasses-screen WIN32 native/screen/app.cpp native/screen/widget.cpp native/screen/glasses_case.cpp native/screen/pixel_controls.cpp
        native/screen/cloth_menu.cpp native/screen/capture.cpp native/screen/diagnostics.cpp native/screen/single_instance.cpp native/screen/app.rc native/screen/app.manifest "${STRINGS_HEADER}")
    set_source_files_properties(native/screen/app.rc PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/screen/assets/Sunglasses.ico")
    set_target_properties(sunglasses-screen PROPERTIES OUTPUT_NAME Sunglasses)
    target_compile_features(sunglasses-screen PRIVATE cxx_std_20)
    target_compile_definitions(sunglasses-screen PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_include_directories(sunglasses-screen PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/native/screen")
    target_link_libraries(sunglasses-screen PRIVATE sunglasses-screen-core dxgi dcomp windowsapp dwmapi wtsapi32 user32 gdi32 comctl32 ole32 shell32 uxtheme uuid advapi32)
    if(MSVC)
        target_compile_options(sunglasses-screen-core PRIVATE /W4 /WX /utf-8 /EHsc)
        target_compile_options(sunglasses-screen PRIVATE /W4 /WX /utf-8 /EHsc)
    endif()
endif()
